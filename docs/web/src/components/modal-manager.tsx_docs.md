export interface IModalManagerChildrenProps {
  showModal(): void;
  hideModal(): void;
  visible: boolean;
}

  children: (props: IModalManagerChildrenProps) => React.ReactNode;
}

const ModalManager = ({ children }: IProps) => {
  const [visible, setVisible] = useState(false);

  const showModal = () => {
    setVisible(true);
  };
  const hideModal = () => {
    setVisible(false);

};
