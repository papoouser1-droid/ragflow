url=$1
timeout=${2:-15}
quiet=${3:-0}

for i in $(seq "$timeout"); do
  if curl -fs "$url" >/dev/null; then
    [[ "$quiet" -ne 1 ]] && echo "✔ $url is healthy after $i seconds"
    exit 0
  fi
  sleep 1
done

echo "✖ Timeout after $timeout seconds waiting for $url"
exit 1
