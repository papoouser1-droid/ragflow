host=$1
port=$2
shift 2

timeout=15
quiet=0

while [[ $# -gt 0 ]]; do
  case "$1" in
  -t | --timeout)
    timeout="$2"
    shift 2
    ;;
  -q | --quiet)
    quiet=1
    shift
    ;;
  *)
    break
    ;;
  esac
done

for i in $(seq "$timeout"); do
  if nc -z "$host" "$port" >/dev/null 2>&1; then
    [[ "$quiet" -ne 1 ]] && echo "✔ $host:$port is available after $i seconds"
    exit 0
  fi
  sleep 1
done

echo "✖ Timeout after $timeout seconds waiting for $host:$port"
exit 1
