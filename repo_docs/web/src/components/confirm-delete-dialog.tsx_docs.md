interface IProps {
  title?: string;

  hidden?: boolean;
}

export function ConfirmDeleteDialog({
  children,
  title,
  onOk,
  onCancel,
  hidden = false,
  onOpenChange,
  open,
  defaultOpen,
}: IProps & DialogProps) {
  const { t } = useTranslation();

  if (hidden) {

            {t('common.no')}
